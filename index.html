Hướng dẫn tích hợp vào app (Swift)

Thêm đoạn code này vào Swift app để gọi API key validation:

// LicenseManager.swift import Foundation struct LicenseManager { static let apiURL = "https://us-central1-nminhiosvn.cloudfunctions.net/validateKey" static func validate(key: String, deviceId: String, completion: @escaping (Result<LicenseInfo, LicenseError>) -> Void) { guard let url = URL(string: apiURL) else { return } var req = URLRequest(url: url) req.httpMethod = "POST" req.setValue("application/json", forHTTPHeaderField: "Content-Type") let body: [String: Any] = [ "key": key, "deviceId": deviceId, "deviceName": UIDevice.current.name ] req.httpBody = try? JSONSerialization.data(withJSONObject: body) URLSession.shared.dataTask(with: req) { data, _, _ in guard let data = data, let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any], let status = json["status"] as? String else { completion(.failure(.network)); return } if status == "ok" { let info = LicenseInfo( keyLabel: json["keyLabel"] as? String ?? key, note: json["note"] as? String ?? "", remainingDays: json["remainingDays"] as? Int, remainingHours: json["remainingHours"] as? Int ) completion(.success(info)) } else { let code = json["code"] as? String ?? "unknown" completion(.failure(LicenseError(rawCode: code))) } }.resume() } } struct LicenseInfo { let keyLabel: String let note: String let remainingDays: Int? let remainingHours: Int? } enum LicenseError: Error { case network, notFound, expired, revoked, deviceLimitReached, deviceMismatch, maintenance(String) init(rawCode: String) { switch rawCode { case "not_found": self = .notFound case "expired": self = .expired case "revoked": self = .revoked case "device_limit_reached": self = .deviceLimitReached case "device_mismatch": self = .deviceMismatch case "maintenance": self = .maintenance("Under Maintenance") default: self = .network } } }
